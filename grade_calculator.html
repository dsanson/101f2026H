Grade Calculator

This doesn’t work quite right. It doesn’t distinguish minimum cut-off values from the values that should be assigned to letter grades.

gradeValues = new Map([
  ["A", 93],
  ["A-", 90],
  ["B+", 87],
  ["B", 83],
  ["B-", 80],
  ["C+", 77],
  ["C", 73],
  ["C-", 70],
  ["D+", 67],
  ["D", 63],
  ["D-", 60],
  ["F", 0]
])

gradeOptionsWithNull = [null, "A", "A-", "B+", "B", "B-", "C+", "C", "C-", "D+", "D", "D-", "F"]

viewof gradeInputs = Inputs.form({
  assignment1: Inputs.select(gradeOptionsWithNull, { label: "Exercises", value: null }),
  assignment2: Inputs.select(gradeOptionsWithNull, { label: "Midterm 1", value: null }),
  assignment3: Inputs.select(gradeOptionsWithNull, { label: "Midterm 2", value: null }),
  assignment4: Inputs.select(gradeOptionsWithNull, { label: "Final Exam", value: null })
})

gradesArray = [
  gradeInputs.assignment1,
  gradeInputs.assignment2,
  gradeInputs.assignment3,
  gradeInputs.assignment4
]

// Average of grades entered so far
averagePercentToDate = {
  const validGrades = gradesArray
    .filter(grade => grade !== null && grade !== "")
    .map(grade => gradeValues.get(grade));
  
  if (validGrades.length === 0) return 0;
  
  const sum = validGrades.reduce((a, b) => a + b, 0);
  return (sum / validGrades.length).toFixed(1);
}

// Average assuming remaining grades are 0
averagePercentFinal = {
  const validGrades = gradesArray
    .filter(grade => grade !== null && grade !== "")
    .map(grade => gradeValues.get(grade));
  
  const sum = validGrades.reduce((a, b) => a + b, 0);
  const average = sum / 4; // Always divide by 4
  return average.toFixed(1);
}

letterGradeToDate = {
  const percent = parseFloat(averagePercentToDate);
  
  if (percent >= 90) return "A-";
  if (percent >= 80) return "B-";
  if (percent >= 70) return "C-";
  if (percent >= 60) return "D-";
  if (percent > 0) return "F";
  return "—";
}

letterGradeFinal = {
  const percent = parseFloat(averagePercentFinal);
  
  if (percent >= 90) return "A-";
  if (percent >= 80) return "B-";
  if (percent >= 70) return "C-";
  if (percent >= 60) return "D-";
  if (percent > 0) return "F";
  return "F";
}
